==================================================================
== REDIS
==================================================================
-- Проникновение в конетйнер
docker exec -it redis_auth bash

redis-cli

-- Ввод секретного ключа
AUTH 12345

-- Просмотр всех ключей
KEYS *

-- Получение значения по ключу
GET <ключ>
GET user:16
HGETALL "sessions:48:462f1ebd-40a8-4ee7-968c-4e2b9efb30ee"


-- Получить все ключи и значения в Redis (если база данных не слишком большая)
HGETALL <ключ>

-- Автоматическое удаление истекших ключей 1 раз в секунду
CONFIG SET hz 1

-- Полная очистка Redis
FLUSHALL ASYNC
FLUSHALL — удаляет все ключи из всех баз данных в Redis.
FLUSHDB — удаляет только ключи из текущей выбранной базы данных.
